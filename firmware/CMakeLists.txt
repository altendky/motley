cmake_minimum_required(VERSION 3.22)

if(NOT DEFINED MOTLEY_BUILD_VERSION OR NOT MOTLEY_BUILD_VERSION MATCHES "^g[0-9a-f]+-(clean|dirty)$")
    message(FATAL_ERROR "Build through scripts/firmware.py to supply source identity")
endif()
string(REGEX REPLACE "^g([0-9a-f]+)-(clean|dirty)$" "\\1" MOTLEY_COMMIT "${MOTLEY_BUILD_VERSION}")
string(LENGTH "${MOTLEY_COMMIT}" MOTLEY_COMMIT_LENGTH)
if(NOT MOTLEY_COMMIT_LENGTH EQUAL 40 AND NOT MOTLEY_COMMIT_LENGTH EQUAL 64)
    message(FATAL_ERROR "Build identity requires the full Git commit hash")
endif()
string(SUBSTRING "${MOTLEY_COMMIT}" 0 12 MOTLEY_SDK_COMMIT)
string(REGEX REPLACE "^g[0-9a-f]+-(clean|dirty)$" "\\1" MOTLEY_WORKING_TREE "${MOTLEY_BUILD_VERSION}")
set(PROJECT_VER "g${MOTLEY_SDK_COMMIT}-${MOTLEY_WORKING_TREE}")
set(PROJECT_VER_NUMBER 1)
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK "/build/dependencies.lock")
project(motley)
